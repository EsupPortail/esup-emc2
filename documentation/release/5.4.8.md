Modification des données pour suivre le changement de séparateur
```postgresql
-- Declaration de la fonction de decodage
CREATE OR REPLACE FUNCTION decode_html_entities(text)
    RETURNS text AS $$
DECLARE
    result text := $1;
BEGIN
    -- Entités françaises courantes
    result := REPLACE(result, '&eacute;', 'é');
    result := REPLACE(result, '&Eacute;', 'É');
    result := REPLACE(result, '&egrave;', 'è');
    result := REPLACE(result, '&Egrave;', 'È');
    result := REPLACE(result, '&ecirc;', 'ê');
    result := REPLACE(result, '&Ecirc;', 'Ê');
    result := REPLACE(result, '&agrave;', 'à');
    result := REPLACE(result, '&Agrave;', 'À');
    result := REPLACE(result, '&acirc;', 'â');
    result := REPLACE(result, '&Acirc;', 'Â');
    result := REPLACE(result, '&ugrave;', 'ù');
    result := REPLACE(result, '&Ugrave;', 'Ù');
    result := REPLACE(result, '&ucirc;', 'û');
    result := REPLACE(result, '&Ucirc;', 'Û');
    result := REPLACE(result, '&ocirc;', 'ô');
    result := REPLACE(result, '&Ocirc;', 'Ô');
    result := REPLACE(result, '&ccedil;', 'ç');
    result := REPLACE(result, '&Ccedil;', 'Ç');

    -- Entités courantes
    result := REPLACE(result, '&amp;', '&');
    result := REPLACE(result, '&lt;', '<');
    result := REPLACE(result, '&gt;', '>');
    result := REPLACE(result, '&quot;', '"');
    result := REPLACE(result, '&apos;', '''');
    result := REPLACE(result, '&nbsp;', ' ');

    RETURN result;
END;
$$ LANGUAGE plpgsql;

-- Fait le changement
update unicaen_autoform_formulaire_reponse set reponse = REGEXP_REPLACE(decode_html_entities(reponse), ';', '###','g')
where
champ in (select c.id
    from unicaen_autoform_champ c
             join unicaen_autoform_categorie a on c.categorie = a.id
             join unicaen_autoform_formulaire f on a.formulaire = f.id
             join unicaen_autoform_champ_type t on c.element = t.code
    where t.code='Plus Custom'
);
```
